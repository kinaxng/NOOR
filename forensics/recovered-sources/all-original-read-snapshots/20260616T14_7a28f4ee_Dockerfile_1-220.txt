# =============================================================================
# NOOR Dockerfile
# - smoke : CPU-only smoke / CI validation image
# - full  : GPU-first production image (CUDA base + Python 3.12 + cu128 torch)
# =============================================================================

ARG NOOR_DOCKER_PROFILE=full
ARG FULL_RUNTIME_IMAGE=nvidia/cuda:12.8.1-runtime-ubuntu24.04

FROM ubuntu:24.04 AS smoke-runtime-base
ARG APT_MIRROR=http://mirrors.tuna.tsinghua.edu.cn
ENV DEBIAN_FRONTEND=noninteractive
ENV PYTHONDONTWRITEBYTECODE=1
ENV PIP_NO_CACHE_DIR=1
ENV PIP_DISABLE_PIP_VERSION_CHECK=1
ENV PIP_DEFAULT_TIMEOUT=1000
ENV VIRTUAL_ENV=/opt/venv
ENV PATH=/opt/venv/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

RUN if [ -f /etc/apt/sources.list.d/ubuntu.sources ]; then \
        sed -i "s|http://archive.ubuntu.com/ubuntu|${APT_MIRROR}/ubuntu|g; s|http://security.ubuntu.com/ubuntu|${APT_MIRROR}/ubuntu|g" /etc/apt/sources.list.d/ubuntu.sources; \
    elif [ -f /etc/apt/sources.list ]; then \
        sed -i "s|http://archive.ubuntu.com/ubuntu|${APT_MIRROR}/ubuntu|g; s|http://security.ubuntu.com/ubuntu|${APT_MIRROR}/ubuntu|g" /etc/apt/sources.list; \
    fi \
    && apt-get update \
    && apt-get install -y --no-install-recommends \
        python3 python3-venv \
        ffmpeg libsm6 libxext6 libgl1 libglib2.0-0 \
        git curl \
    && rm -rf /var/lib/apt/lists/*

FROM ${FULL_RUNTIME_IMAGE} AS full-runtime-base
ARG APT_MIRROR=http://mirrors.tuna.tsinghua.edu.cn
ENV DEBIAN_FRONTEND=noninteractive
ENV PYTHONDONTWRITEBYTECODE=1
ENV PIP_NO_CACHE_DIR=1
ENV PIP_DISABLE_PIP_VERSION_CHECK=1
ENV PIP_DEFAULT_TIMEOUT=1000
ENV VIRTUAL_ENV=/opt/venv

RUN if [ -f /etc/apt/sources.list.d/ubuntu.sources ]; then \
        sed -i "s|http://archive.ubuntu.com/ubuntu|${APT_MIRROR}/ubuntu|g; s|http://security.ubuntu.com/ubuntu|${APT_MIRROR}/ubuntu|g" /etc/apt/sources.list.d/ubuntu.sources; \
    elif [ -f /etc/apt/sources.list ]; then \
        sed -i "s|http://archive.ubuntu.com/ubuntu|${APT_MIRROR}/ubuntu|g; s|http://security.ubuntu.com/ubuntu|${APT_MIRROR}/ubuntu|g" /etc/apt/sources.list; \
    fi \
    && apt-get update \
    && apt-get install -y --no-install-recommends \
        python3 python3-venv \
        ffmpeg libsm6 libxext6 libgl1 libglib2.0-0 \
        git curl \
    && rm -rf /var/lib/apt/lists/*

FROM smoke-runtime-base AS smoke-builder
ARG PIP_INDEX_URL=https://pypi.tuna.tsinghua.edu.cn/simple
ARG PYTORCH_WHEEL_MIRROR=https://mirrors.aliyun.com/pytorch-wheels
ARG INSTALL_NEMO=false
ENV PIP_INDEX_URL=${PIP_INDEX_URL}

RUN apt-get update \
    && apt-get install -y --no-install-recommends \
        python3-pip wget build-essential \
    && rm -rf /var/lib/apt/lists/* \
    && python3 -m venv /opt/venv \
    && /opt/venv/bin/python -m pip install --upgrade pip setuptools wheel \
    && ln -sf /usr/bin/python3 /usr/local/bin/python

RUN wget -nv --tries=10 --waitretry=5 -O /tmp/torch-2.8.0+cpu-cp312-cp312-manylinux_2_28_x86_64.whl \
        "${PYTORCH_WHEEL_MIRROR}/cpu/torch-2.8.0%2Bcpu-cp312-cp312-manylinux_2_28_x86_64.whl" \
    && wget -nv --tries=10 --waitretry=5 -O /tmp/torchvision-0.23.0+cpu-cp312-cp312-manylinux_2_28_x86_64.whl \
        "${PYTORCH_WHEEL_MIRROR}/cpu/torchvision-0.23.0%2Bcpu-cp312-cp312-manylinux_2_28_x86_64.whl" \
    && python3 -m pip install --retries 10 \
        /tmp/torch-2.8.0+cpu-cp312-cp312-manylinux_2_28_x86_64.whl \
        /tmp/torchvision-0.23.0+cpu-cp312-cp312-manylinux_2_28_x86_64.whl \
    && rm -f \
        /tmp/torch-2.8.0+cpu-cp312-cp312-manylinux_2_28_x86_64.whl \
        /tmp/torchvision-0.23.0+cpu-cp312-cp312-manylinux_2_28_x86_64.whl

COPY backend/app/pipeline/lada/ /app/lada/
RUN python3 -m pip install --retries 10 /app/lada[docker]

RUN python3 -m pip install \
    --retries 10 \
    "faster-whisper>=1.0.0" \
    "transformers>=4.35.0" \
    "accelerate>=0.25.0"

RUN if [ "$INSTALL_NEMO" = "true" ]; then \
        python3 -m pip install --retries 10 "nemo_toolkit[asr]==2.7.3"; \
    else \
        echo "Skipping optional nemo_toolkit[asr]"; \
    fi

COPY backend/requirements.docker.txt /app/requirements.docker.txt
RUN python3 -m pip install --retries 10 -r /app/requirements.docker.txt

        "${PYTORCH_WHEEL_MIRROR}/cu128/torch-2.8.0%2Bcu128-cp312-cp312-manylinux_2_28_x86_64.whl" \
    && wget -nv --tries=10 --waitretry=5 -O /tmp/torchvision-0.23.0+cu128-cp312-cp312-manylinux_2_28_x86_64.whl \
        "${PYTORCH_WHEEL_MIRROR}/cu128/torchvision-0.23.0%2Bcu128-cp312-cp312-manylinux_2_28_x86_64.whl" \
    && /opt/venv/bin/python -m pip install --retries 10 \
        /tmp/torch-2.8.0+cu128-cp312-cp312-manylinux_2_28_x86_64.whl \
        /tmp/torchvision-0.23.0+cu128-cp312-cp312-manylinux_2_28_x86_64.whl \

        /tmp/torch-2.8.0+cu128-cp312-cp312-manylinux_2_28_x86_64.whl \
        /tmp/torchvision-0.23.0+cu128-cp312-cp312-manylinux_2_28_x86_64.whl

COPY backend/app/pipeline/lada/ /app/lada/
RUN /opt/venv/bin/python -m pip install --retries 10 /app/lada[nvidia,docker]

RUN /opt/venv/bin/python -m pip install \
    --retries 10 \
    "faster-whisper>=1.0.0" \
    "transformers>=4.35.0" \
    "accelerate>=0.25.0"

RUN if [ "$INSTALL_NEMO" = "true" ]; then \
        /opt/venv/bin/python -m pip install --retries 10 "nemo_toolkit[asr]==2.7.3"; \
    else \
        echo "Skipping optional nemo_toolkit[asr]"; \
    fi

COPY backend/requirements.docker.txt /app/requirements.docker.txt
RUN /opt/venv/bin/python -m pip install --retries 10 -r /app/requirements.docker.txt

FROM smoke-runtime-base AS smoke-final
ENV LADA_MODEL_WEIGHTS_DIR=/app/models/lada_model_weights
ENV AUDIO_SEPARATOR_MODEL_DIR=/app/models/audio-separator
ENV HF_HOME=/app/models/huggingface
ENV TMPDIR=/app/tmp
ENV XDG_CACHE_HOME=/app/cache
ENV TRANSFORMERS_CACHE=/app/cache/huggingface
ENV TORCH_HOME=/app/cache/torch
ENV MPLCONFIGDIR=/app/cache/matplotlib
WORKDIR /app
COPY --from=smoke-builder /opt/venv /opt/venv
COPY backend/app /app/app
COPY backend/run.py /app/
COPY .env.example /app/.env.example
RUN mkdir -p /app/models/lada_model_weights /app/models/huggingface /app/tmp
EXPOSE 9898
CMD ["/opt/venv/bin/python", "/app/run.py"]
