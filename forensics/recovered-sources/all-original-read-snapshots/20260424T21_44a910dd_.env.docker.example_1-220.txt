# docker-compose.yml 中通过 volumes 挂载了 ./models 到 /app/models
# 所以这里路径指向容器内路径

# --- LADA ---
LADA_CLI_PATH=/usr/local/bin/lada-cli
LADA_MODEL_WEIGHTS_DIR=/app/models/lada_model_weights

# --- Whisper ---
HF_HOME=/app/models/huggingface
HF_TOKEN=                        # HuggingFace token（gated repos 需要）

# --- 媒体路径（容器内 /Videos，映射自宿主机 /home/kinax/Videos） ---
SOURCE_DIR=/Videos
OUTPUT_DIR=/Videos

# --- Emby ---
EMBY_SERVER=http://localhost:8096
EMBY_API_KEY=your_emby_api_key_here

# --- 网络加速（国内推荐 mirror） ---
ACCELERATION_MODE=mirror
HF_ENDPOINT=https://hf-mirror.com
PIP_MIRROR=https://pypi.tuna.tsinghua.edu.cn/simple

# --- 服务器 ---
HOST=0.0.0.0
PORT=9898
RELOAD=false
