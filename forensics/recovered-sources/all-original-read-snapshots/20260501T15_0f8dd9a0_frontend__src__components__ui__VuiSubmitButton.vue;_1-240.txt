<script setup lang="ts">
import { ref, watch, nextTick, onMounted, onUnmounted } from 'vue'

const props = withDefaults(defineProps<{
  tabs: { key: string; label: string }[]
  modelValue: string
  compact?: boolean
}>(), {
  compact: false,
})

const emit = defineEmits<{
  (e: 'update:modelValue', value: string): void
}>()

const tabRefs = ref<HTMLElement[]>([])
const containerRef = ref<HTMLElement | null>(null)
const indicator = ref({ left: 0, width: 0 })

function updateIndicator() {
  nextTick(() => {
    const activeIdx = props.tabs.findIndex(t => t.key === props.modelValue)
    if (activeIdx < 0 || !containerRef.value) return
    const tabEl = tabRefs.value[activeIdx]
    if (!tabEl) return
    const containerRect = containerRef.value.getBoundingClientRect()
    const tabRect = tabEl.getBoundingClientRect()
    indicator.value = {
      left: tabRect.left - containerRect.left,
      width: tabRect.width,
    }
  })
}

watch(() => props.modelValue, updateIndicator)
watch(() => props.tabs, () => nextTick(updateIndicator), { deep: true })

onMounted(() => {
  updateIndicator()
  const ro = new ResizeObserver(() => updateIndicator())
  if (containerRef.value) ro.observe(containerRef.value)
  onUnmounted(() => ro.disconnect())
})

function selectTab(key: string) {
  emit('update:modelValue', key)
}
</script>

<template>
  <div class="vision-tabs" :class="{ 'vision-tabs--compact': props.compact }" ref="containerRef" style="max-width: 100%; overflow-x: auto;">
    <!-- Sliding indicator -->
    <div

      :style="{
        left: indicator.left + 'px',
        width: indicator.width + 'px',
      }"
    />
    <!-- Tab buttons -->
    <button
      v-for="(tab, idx) in tabs"
      :key="tab.key"
      :ref="el => { if (el) tabRefs[idx] = el as HTMLElement }"
      class="vision-tabs__tab"
      :class="{ 'vision-tabs__tab--active': modelValue === tab.key }"
      @click="selectTab(tab.key)"
    >
      {{ tab.label }}
    </button>
  </div>
</template>

--- sdkUi ---
type Child = Node | string | number | null | undefined | false

type ButtonTone = 'default' | 'primary' | 'danger' | 'ghost'
type SubmitStatus = 'idle' | 'submitting' | 'success' | 'error'
type NoticeTone = 'info' | 'success' | 'error' | 'warning'

function appendChildren(el: HTMLElement, children?: Child | Child[]) {
  const list = Array.isArray(children) ? children : [children]
  for (const child of list) {
    if (child === null || child === undefined || child === false) continue
    if (child instanceof Node) el.appendChild(child)
    else el.appendChild(document.createTextNode(String(child)))
  }
}

function setClass(el: HTMLElement, base: string, extra?: string) {
  el.className = extra ? `${base} ${extra}` : base
}

function isTypingTarget(target: EventTarget | null) {
  const el = target as HTMLElement | null
  const tag = el?.tagName?.toLowerCase() || ''
  return tag === 'input' || tag === 'textarea' || tag === 'select' || !!el?.isContentEditable
}

function clampPage(page: number, pages: number) {
  return Math.max(1, Math.min(Math.trunc(page || 1), Math.max(1, pages)))
}

function pageItems(page: number, pages: number, siblingCount = 1): (number | 'ellipsis')[] {
  const total = Math.max(1, pages)
  const current = clampPage(page, total)
  const siblings = Math.max(0, siblingCount)
  const maxItems = 5 + siblings * 2
  if (total <= maxItems) return Array.from({ length: total }, (_, idx) => idx + 1)
  const left = Math.max(2, current - siblings)
  const right = Math.min(total - 1, current + siblings)
  const items: (number | 'ellipsis')[] = [1]
  if (left > 2) items.push('ellipsis')
  for (let p = left; p <= right; p += 1) items.push(p)
  if (right < total - 1) items.push('ellipsis')
  items.push(total)
  return items
}

export function createPluginUi() {
  function page(options: { className?: string; children?: Child | Child[] } = {}) {
    const el = document.createElement('div')
    setClass(el, 'noor-plugin-page', options.className)
    appendChildren(el, options.children)
    return el
  }

  function button(options: {
    label?: string
    text?: string
    tone?: ButtonTone
    active?: boolean
    disabled?: boolean
    className?: string
    title?: string
    onClick?: (event: MouseEvent) => void
  } = {}) {
    const el = document.createElement('button')
    el.type = 'button'
    el.textContent = options.label ?? options.text ?? ''
    setClass(el, `noor-plugin-btn noor-plugin-btn--${options.tone || 'default'}${options.active ? ' is-active' : ''}`, options.className)
    el.disabled = !!options.disabled
    if (options.title) el.title = options.title
    if (options.onClick) el.addEventListener('click', options.onClick)
    return el
  }

  function submitButton(options: {
    label?: string
    idleLabel?: string
    submittingLabel?: string
    successLabel?: string
    errorLabel?: string
    status?: SubmitStatus
    progress?: number
    disabled?: boolean
    className?: string
    onClick?: (event: MouseEvent) => void
  } = {}) {
    const el = document.createElement('button')
    el.type = 'button'
    const status = options.status || 'idle'
    const progress = Math.max(0, Math.min(100, status === 'success' ? 100 : options.progress || 0))
    setClass(el, `noor-plugin-submit noor-plugin-submit--${status}`, options.className)
    el.disabled = !!options.disabled || status === 'submitting' || status === 'success'
    el.innerHTML = `${status === 'submitting' || status === 'success' ? `<i style="width:${progress}%"></i>` : ''}<span></span>`
    const label = el.querySelector('span') as HTMLSpanElement
    label.textContent = status === 'submitting'
      ? (options.submittingLabel || (progress ? `${Math.round(progress)}%` : '提交中'))
      : status === 'success'
        ? (options.successLabel || '已提交')
        : status === 'error'
          ? (options.errorLabel || '提交失败')
          : (options.label || options.idleLabel || '提交')
    if (options.onClick) el.addEventListener('click', event => {
      if (el.disabled) return
      options.onClick?.(event)
    })
    return el
  }

  function badge(options: { label?: string; text?: string; active?: boolean; tone?: NoticeTone | 'muted'; className?: string; onClick?: (event: MouseEvent) => void } = {}) {
    const el = document.createElement(options.onClick ? 'button' : 'span') as HTMLElement
    if (el instanceof HTMLButtonElement) el.type = 'button'
    el.textContent = options.label ?? options.text ?? ''
    setClass(el, `noor-plugin-badge noor-plugin-badge--${options.tone || 'muted'}${options.active ? ' is-active' : ''}`, options.className)
    if (options.onClick) el.addEventListener('click', options.onClick)
    return el
  }

  function tabs(options: { value?: string; tabs: Array<{ key: string; label: string; disabled?: boolean }>; className?: string; onChange?: (key: string) => void }) {
    const wrap = document.createElement('div')
    setClass(wrap, 'noor-plugin-tabs', options.className)
    const indicator = document.createElement('div')
    indicator.className = 'noor-plugin-tabs__indicator'
    wrap.appendChild(indicator)

    let indicatorReady = false
    const updateIndicator = () => {
      const active = wrap.querySelector<HTMLButtonElement>('.noor-plugin-tab.is-active')
      if (!active) { indicator.style.opacity = '0'; return }
      const wrapRect = wrap.getBoundingClientRect()
      const activeRect = active.getBoundingClientRect()
      if (!indicatorReady) indicator.style.transition = 'none'
      indicator.style.opacity = '1'
      indicator.style.left = `${activeRect.left - wrapRect.left + wrap.scrollLeft}px`
      indicator.style.width = `${activeRect.width}px`
      if (!indicatorReady) {
        indicatorReady = true
        window.requestAnimationFrame(() => { indicator.style.transition = '' })
      }
    }

    for (const item of options.tabs || []) {
      const tab = document.createElement('button')
      tab.type = 'button'
      tab.className = `noor-plugin-tab${item.key === options.value ? ' is-active' : ''}`
      tab.textContent = item.label
      tab.disabled = !!item.disabled
      tab.dataset.tab = item.key
      tab.onclick = () => {
        if (tab.classList.contains('is-active')) return
        for (const other of wrap.querySelectorAll('.noor-plugin-tab')) other.classList.remove('is-active')
        tab.classList.add('is-active')
        updateIndicator()
        options.onChange?.(item.key)
      }
      wrap.appendChild(tab)
    }

    const ro = typeof ResizeObserver !== 'undefined' ? new ResizeObserver(updateIndicator) : null
    ro?.observe(wrap)
    window.requestAnimationFrame(updateIndicator)
    window.setTimeout(updateIndicator, 0)
    ;(wrap as HTMLElement & { dispose?: () => void; updateIndicator?: () => void }).dispose = () => ro?.disconnect()
    ;(wrap as HTMLElement & { dispose?: () => void; updateIndicator?: () => void }).updateIndicator = updateIndicator
    return wrap
  }

  function pagination(options: {
    page: number
    total?: number
    pageSize?: number
    totalPages?: number
    siblingCount?: number
    className?: string
    onPage?: (page: number) => void
  }) {
    const totalPages = options.totalPages ?? Math.ceil(Math.max(0, options.total || 0) / Math.max(1, options.pageSize || 20))
    const pages = Math.max(1, totalPages || 1)
    const current = clampPage(options.page, pages)
    const nav = document.createElement('nav')
    setClass(nav, 'noor-plugin-pagination', options.className)
    nav.setAttribute('aria-label', 'Pagination')
    const go = (next: number) => {
      const safe = clampPage(next, pages)
      if (safe !== current) options.onPage?.(safe)
    }
    nav.appendChild(button({ label: '上一页', disabled: current <= 1, className: 'noor-plugin-page-btn', onClick: () => go(current - 1) }))
    const pagesWrap = document.createElement('div')
    pagesWrap.className = 'noor-plugin-pagination__pages'
    for (const item of pageItems(current, pages, options.siblingCount)) {
      if (item === 'ellipsis') {
        const ellipsis = document.createElement('span')
        ellipsis.className = 'noor-plugin-pagination__ellipsis'
        ellipsis.textContent = '…'
        pagesWrap.appendChild(ellipsis)
      } else {
        pagesWrap.appendChild(button({ label: String(item), active: item === current, className: 'noor-plugin-page-number', onClick: () => go(item) }))
      }
    }
    nav.appendChild(pagesWrap)
    const compact = document.createElement('span')
    compact.className = 'noor-plugin-pagination__compact'
    compact.textContent = `${current} / ${pages}`
    nav.appendChild(compact)
    nav.appendChild(button({ label: '下一页', disabled: current >= pages, className: 'noor-plugin-page-btn', onClick: () => go(current + 1) }))
    const onKeydown = (event: KeyboardEvent) => {
      if (isTypingTarget(event.target) || event.altKey || event.ctrlKey || event.metaKey || event.shiftKey) return
      if (event.key === 'Home') { event.preventDefault(); go(1) }
      else if (event.key === 'End') { event.preventDefault(); go(pages) }
      else if (event.key === 'PageUp') { event.preventDefault(); go(current - 1) }
      else if (event.key === 'PageDown') { event.preventDefault(); go(current + 1) }
    }
    window.addEventListener('keydown', onKeydown)
    ;(nav as HTMLElement & { dispose?: () => void }).dispose = () => window.removeEventListener('keydown', onKeydown)
    return nav
  }

  function notice(options: { text?: string; tone?: NoticeTone; className?: string } = {}) {
    const el = document.createElement('div')
    el.textContent = options.text || ''
    setClass(el, `noor-plugin-notice noor-plugin-notice--${options.tone || 'info'}`, options.className)
    return el
  }

  function emptyState(options: { text?: string; className?: string } = {}) {
    const el = document.createElement('div')
    el.textContent = options.text || '暂无内容'
    setClass(el, 'noor-plugin-empty', options.className)
    return el
  }

  function chip(options: {
    label?: string
    text?: string
    active?: boolean
    disabled?: boolean
    className?: string
    onClick?: (event: MouseEvent) => void
  } = {}) {
    const el = document.createElement(options.onClick ? 'button' : 'span') as HTMLElement
    if (el instanceof HTMLButtonElement) el.type = 'button'
    el.textContent = options.label ?? options.text ?? ''
    setClass(el, `noor-plugin-chip${options.active ? ' is-active' : ''}`, options.className)
    if (el instanceof HTMLButtonElement) el.disabled = !!options.disabled
    if (options.onClick) el.addEventListener('click', options.onClick)
    return el
  }

  function card(options: {
    children?: Child | Child[]
    className?: string
    href?: string
    target?: string
    onClick?: (event: MouseEvent) => void
  } = {}) {
    const el = document.createElement(options.href ? 'a' : 'div') as HTMLElement
    setClass(el, 'noor-plugin-card', options.className)
    if (options.href && el instanceof HTMLAnchorElement) {
      el.href = options.href
      el.target = options.target || '_self'
      if (el.target === '_blank') el.rel = 'noopener noreferrer'
    }
