# =============================================================================
# NOOR Dockerfile
# - full  : CUDA 12.8 运行镜像
# - smoke : 纯 CPU / 轻量基底，仅用于 CI / 本地 smoke build 验证
# =============================================================================
# 用法:
#   docker build -t noor-backend .
#   docker build --build-arg NOOR_DOCKER_PROFILE=smoke -t noor-backend-smoke .
# =============================================================================

ARG NOOR_DOCKER_PROFILE=full

FROM ubuntu:24.04 AS smoke-base
ENV DEBIAN_FRONTEND=noninteractive

FROM nvidia/cuda:12.8.0-runtime-ubuntu24.04 AS full-base
ENV DEBIAN_FRONTEND=noninteractive

FROM ${NOOR_DOCKER_PROFILE}-base

ARG NOOR_DOCKER_PROFILE
ARG APT_MIRROR=http://mirrors.tuna.tsinghua.edu.cn
ARG PIP_INDEX_URL=https://pypi.tuna.tsinghua.edu.cn/simple

ENV PYTHONDONTWRITEBYTECODE=1
ENV PIP_NO_CACHE_DIR=1
ENV PIP_DISABLE_PIP_VERSION_CHECK=1
ENV PIP_DEFAULT_TIMEOUT=1000
ENV PIP_INDEX_URL=${PIP_INDEX_URL}

# ---------------------------------------------------------------------------
# 1. 系统依赖
# ---------------------------------------------------------------------------
RUN sed -i "s|http://archive.ubuntu.com/ubuntu|${APT_MIRROR}/ubuntu|g; s|http://security.ubuntu.com/ubuntu|${APT_MIRROR}/ubuntu|g" /etc/apt/sources.list.d/ubuntu.sources \
    && apt-get update && apt-get install -y --no-install-recommends \
    python3 python3-venv python3-pip \
    ffmpeg libsm6 libxext6 libgl1 libglib2.0-0 \
    git curl wget build-essential \
    && rm -rf /var/lib/apt/lists/* \
    && ln -sf /usr/bin/python3 /usr/local/bin/python

# ---------------------------------------------------------------------------
# 2. PyTorch / torchvision
#    - full: CUDA 12.8 wheels
#    - smoke: CPU wheels（仅用于构建验证）
# ---------------------------------------------------------------------------
RUN if [ "$NOOR_DOCKER_PROFILE" = "smoke" ]; then \
        python3 -m pip install \
            --retries 10 \
            "torch==2.8.0" \
            "torchvision==0.23.0" \
            --index-url https://download.pytorch.org/whl/cpu ; \
    else \
        python3 -m pip install \
            --retries 10 \
            "torch==2.8.0" \
            "torchvision==0.23.0" \
            --index-url https://download.pytorch.org/whl/cu128 ; \
    fi

# ---------------------------------------------------------------------------
# 3. lada (包含 lada-cli)
#    - 来自 backend/app/pipeline/lada/ (editable install)
#    - extras: nvidia (torch cu128) + docker (opencv-python-headless)
# ---------------------------------------------------------------------------
COPY backend/app/pipeline/lada/ /app/lada/
RUN if [ "$NOOR_DOCKER_PROFILE" = "smoke" ]; then \
        python3 -m pip install --retries 10 -e /app/lada[docker] ; \
    else \
        python3 -m pip install --retries 10 -e /app/lada[nvidia,docker] ; \
    fi

# ---------------------------------------------------------------------------
# 4. Whisper 依赖
#    - faster-whisper: CTranslate2 GPU 加速
#    - transformers + accelerate: anime-whisper / KotobaWhisper
# ---------------------------------------------------------------------------
RUN python3 -m pip install \
    --retries 10 \
    "faster-whisper>=1.0.0" \
    "transformers>=4.35.0" \
    "accelerate>=0.25.0"

# ---------------------------------------------------------------------------
# 5. WebUI 后端依赖
# ---------------------------------------------------------------------------
COPY backend/requirements.txt /app/requirements.txt
RUN python3 -m pip install --retries 10 -r /app/requirements.txt

# ---------------------------------------------------------------------------
# 6. WebUI 应用代码
# ---------------------------------------------------------------------------
COPY backend/app /app/app
COPY backend/run.py /app/
COPY .env.example /app/.env.example

# ---------------------------------------------------------------------------
# 7. 模型目录 (运行时由 docker-compose.yml 覆盖)
#    - /app/models/lada_model_weights/  → lada 检测/修复模型
#    - /app/models/huggingface/          → Whisper 模型缓存
# ---------------------------------------------------------------------------
ENV LADA_MODEL_WEIGHTS_DIR=/app/models/lada_model_weights
ENV HF_HOME=/app/models/huggingface
ENV TMPDIR=/app/tmp
WORKDIR /app

RUN mkdir -p /app/models/lada_model_weights /app/models/huggingface /app/tmp

# ---------------------------------------------------------------------------
# 8. 端口与启动命令
# ---------------------------------------------------------------------------
EXPOSE 9898
CMD ["python3", "/app/run.py"]
---SESSIONS---
