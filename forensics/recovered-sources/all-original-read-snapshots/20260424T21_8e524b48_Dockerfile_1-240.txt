# =============================================================================
# NOOR Dockerfile — CUDA 12.8 (通用版，适配 560/575/590+ Driver)
# =============================================================================
# 用法:
#   docker build -t lada-webui .
#   docker-compose up -d
#
# 前置要求:
#   - NVIDIA Driver 560+ (Linux)
#   - nvidia-container-toolkit
#     https://docs.nvidia.com/datacenter/cloud-native/container-toolkit/install-guide.html
#
# GPU 验证:
#   docker run --rm --gpus all nvidia/cuda:12.8.0-runtime-ubuntu22.04 nvidia-smi
# =============================================================================

FROM nvidia/cuda:12.8.0-runtime-ubuntu24.04

ENV DEBIAN_FRONTEND=noninteractive
ENV PYTHONDONTWRITEBYTECODE=1
ENV PIP_NO_CACHE_DIR=1
ENV PIP_DISABLE_PIP_VERSION_CHECK=1

# ---------------------------------------------------------------------------
# 1. 系统依赖
# ---------------------------------------------------------------------------
RUN apt-get update && apt-get install -y --no-install-recommends \
    # Python + build tools
    python3 python3-venv python3-pip \
    # 媒体处理
    ffmpeg libsm6 libxext6 libgl1 libglib2.0-0 \
    # 基础工具
    git curl wget build-essential \
    # 清理
    && rm -rf /var/lib/apt/lists/* \
    && ln -sf /usr/bin/python3 /usr/local/bin/python

# ---------------------------------------------------------------------------
# 2. PyTorch (CUDA 12.8) + torchvision
#    - 源索引: download.pytorch.org/whl/cu128
#    - RTX 50 系列 (Driver 590) / RTX 40 系列 (Driver 560/575) 通用
# ---------------------------------------------------------------------------
RUN python3 -m pip install \
    "torch==2.8.0" \
    "torchvision==0.23.0" \
    --index-url https://download.pytorch.org/whl/cu128

# ---------------------------------------------------------------------------
# 3. lada (包含 lada-cli)
#    - 来自 backend/app/pipeline/lada/ (editable install)
#    - extras: nvidia (torch cu128) + docker (opencv-python-headless)
# ---------------------------------------------------------------------------
COPY backend/app/pipeline/lada/ /app/lada/
RUN python3 -m pip install -e /app/lada[nvidia,docker]

# ---------------------------------------------------------------------------
# 4. Whisper 依赖
#    - faster-whisper: CTranslate2 GPU 加速
#    - transformers + accelerate: anime-whisper / KotobaWhisper
# ---------------------------------------------------------------------------
RUN python3 -m pip install \
    "faster-whisper>=1.0.0" \
    "transformers>=4.35.0" \
    "accelerate>=0.25.0"

# ---------------------------------------------------------------------------
# 5. WebUI 后端依赖
# ---------------------------------------------------------------------------
COPY backend/requirements.txt /app/requirements.txt
RUN python3 -m pip install -r /app/requirements.txt

# ---------------------------------------------------------------------------
# 6. WebUI 应用代码
# ---------------------------------------------------------------------------
COPY backend/app /app/app
COPY backend/run.py /app/
COPY .env.example /app/.env.example

# ---------------------------------------------------------------------------
# 7. 模型目录 (运行时由 docker-compose.yml 覆盖)
#    - /app/models/lada_model_weights/  → lada 检测/修复模型
#    - /app/models/huggingface/          → Whisper 模型缓存
# ---------------------------------------------------------------------------
ENV LADA_MODEL_WEIGHTS_DIR=/app/models/lada_model_weights
ENV HF_HOME=/app/models/huggingface
ENV TMPDIR=/app/tmp
WORKDIR /app

# 预创建目录
RUN mkdir -p /app/models/lada_model_weights /app/models/huggingface /app/tmp

# ---------------------------------------------------------------------------
# 8. 端口与启动命令
# ---------------------------------------------------------------------------
EXPOSE 9898
CMD ["python3", "/app/run.py"]
